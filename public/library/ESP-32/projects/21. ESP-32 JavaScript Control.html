JavaScript ESP32 Control

Brooo, Project 20 made our ESP32 generate a webpage containing sensor data. But every refresh downloaded the whole page again. 😭

Today we change the architecture. The webpage will stay exactly where it is while JavaScript quietly asks the ESP32 for tiny pieces of fresh data in the background.

The big upgrade: We are introducing JavaScript into our ESP32 webpage. This gives the browser its own logic and lets it communicate with the ESP32 after the page has already loaded.

06 // SOURCE CODE

Complete Code

CRITICAL: Replace YOUR_WIFI_NAME and YOUR_PASSWORD with your actual Wi-Fi credentials.
C++ · Arduino 
/*
 * ============================================================
 * MY BRO'S TWENTY-FIRST ESP32 PROGRAM: JAVASCRIPT CONTROL
 * ============================================================
 */

#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_PASSWORD";

WebServer server(80);

const int LED_PIN = 2;
const int LDR_PIN = 34;

int ledState = LOW;

// ============================================================
// 1. THE HOMEPAGE
// ============================================================

void handle_Homepage() {

    String html =
        "<!DOCTYPE html>"
        "<html>"
        "<head>"
        "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1\">";

    // ========================================================
    // JAVASCRIPT
    // ========================================================

    html += "<script>";

    // --------------------------------------------------------
    // FUNCTION 1: TOGGLE LED
    // --------------------------------------------------------

    html += "function toggleLed() {";

    html +=
        "fetch('/toggle_led')"
        ".then(response => response.text())"
        ".then(text => {";

    html +=
        "document.getElementById('led_status').innerHTML = text;";

    html +=
        "});";

    html += "}";

    // --------------------------------------------------------
    // FUNCTION 2: UPDATE SENSOR
    // --------------------------------------------------------

    html += "function updateSensor() {";

    html +=
        "fetch('/read_sensor')"
        ".then(response => response.text())"
        ".then(text => {";

    html +=
        "document.getElementById('sensor_val').innerHTML = text;";

    html +=
        "});";

    html += "}";

    // --------------------------------------------------------
    // RUN SENSOR UPDATE EVERY 1 SECOND
    // --------------------------------------------------------

    html +=
        "setInterval(updateSensor, 1000);";

    html += "</script>";

    // ========================================================
    // END JAVASCRIPT
    // ========================================================

    html +=
        "<style>"
        "body {"
        "font-family: sans-serif;"
        "text-align: center;"
        "margin-top: 50px;"
        "}"
        "</style>";

    html +=
        "</head>"
        "<body>";

    // ========================================================
    // PAGE CONTENT
    // ========================================================

    html +=
        "<h1>⚡ Live JS Dashboard</h1>";

    // Sensor value
    html +=
        "<h2>"
        "Light Level: "
        "<span id=\"sensor_val\" "
        "style=\"color:orange;\">"
        "Loading..."
        "</span>"
        "</h2>";

    // LED state
    html +=
        "<h2>"
        "LED Status: "
        "<span id=\"led_status\" "
        "style=\"color:blue;\">"
        "OFF"
        "</span>"
        "</h2>";

    // Button
    html +=
        "<button "
        "onclick=\"toggleLed()\" "
        "style=\""
        "padding:20px;"
        "font-size:20px;"
        "background:black;"
        "color:white;"
        "\">"
        "TOGGLE LIGHT"
        "</button>";

    html +=
        "</body>"
        "</html>";

    // Send the complete page
    server.send(
        200,
        "text/html",
        html
    );
}

// ============================================================
// 2. LED API ENDPOINT
// ============================================================

void handle_Toggle() {

    // Flip the LED state
    if (ledState == LOW) {

        ledState = HIGH;

    }
    else {

        ledState = LOW;

    }

    digitalWrite(
        LED_PIN,
        ledState
    );

    // Return only a tiny piece of text
    if (ledState == HIGH) {

        server.send(
            200,
            "text/plain",
            "ON"
        );

    }
    else {

        server.send(
            200,
            "text/plain",
            "OFF"
        );

    }
}

// ============================================================
// 3. SENSOR API ENDPOINT
// ============================================================

void handle_Sensor() {

    // Read the LDR
    int lightLevel =
        analogRead(LDR_PIN);

    // Return only the raw number
    server.send(
        200,
        "text/plain",
        String(lightLevel)
    );
}

// ============================================================
// SETUP
// ============================================================

void setup() {

    Serial.begin(115200);

    pinMode(
        LED_PIN,
        OUTPUT
    );

    // Start Wi-Fi
    WiFi.begin(
        ssid,
        password
    );

    while (
        WiFi.status() != WL_CONNECTED
    ) {

        delay(500);

        Serial.print(".");
    }

    Serial.println(
        "\n🔥 Wi-Fi Connected!"
    );

    Serial.print(
        "Open this IP: http://"
    );

    Serial.println(
        WiFi.localIP()
    );

    // Register routes
    server.on(
        "/",
        handle_Homepage
    );

    server.on(
        "/toggle_led",
        handle_Toggle
    );

    server.on(
        "/read_sensor",
        handle_Sensor
    );

    // Start web server
    server.begin();
}

// ============================================================
// LOOP
// ============================================================

void loop() {

    server.handleClient();
}
Three routes now exist: / gives the webpage, /toggle_led controls the LED, and /read_sensor returns the LDR value.

RESPONSE

JAVASCRIPT

The browser takes the HTTP response and converts its body into text.

Finding the Element

document
    .getElementById('sensor_val')

UPDATE

Replace the Old Value

JAVASCRIPT 
element.innerHTML = text;

The old value disappears and the new sensor value appears without reloading the page.

SETINTERVAL

Repeat Automatically

JAVASCRIPT 
setInterval(updateSensor, 1000);

JavaScript runs updateSensor() every 1000 milliseconds.

TEXT/PLAIN

Tiny API Responses

C++ 
server.send(
    200,
    "text/plain",
    String(lightLevel)
);

The ESP32 does not rebuild the whole webpage. It returns only the requested sensor value.

Important architecture idea: The webpage and the data are now becoming separate layers.

10 // EXPERIMENT

Make It Update Faster

Right now the browser asks the ESP32 for a new sensor value every one second.

CURRENT 
html += "setInterval(updateSensor, 1000);";

Change 1000 to 100 .

EXPERIMENT 
html += "setInterval(updateSensor, 100);";
1000 ms

1 update / second

Simple, lightweight, and easy to observe.

100 ms

10 updates / second

The sensor value can appear much more responsive.

But don't confuse polling with video streaming. At 100 ms, you're still making repeated HTTP requests. You are not creating a continuous video stream.

Faster polling also means more requests and more work for the ESP32 and Wi-Fi network.
The deeper lesson: Good web engineering is not simply "make it faster." It is about balancing responsiveness, network traffic, CPU work, power consumption, and reliability.

12 // DEBUG

Troubleshooting

JAVASCRIPT

"Loading..." never changes

  • Check the fetch('/read_sensor') URL.
  • Make sure the ESP32 registered /read_sensor .
  • Check the browser's JavaScript syntax.
  • Make sure sensor_val exactly matches the HTML ID.
DOM

LED text doesn't change

The JavaScript and HTML must use the exact same element ID.

HTML 
id="led_status"

And JavaScript must search for:

JAVASCRIPT 
getElementById('led_status')
API

Page loads but API doesn't

Check the endpoint registration.

C++ 
server.on(
    "/read_sensor",
    handle_Sensor
);
LDR

Sensor value is strange

  • Verify LDR → 3V3.
  • Verify 10kΩ → GND.
  • Verify the shared row → D34.
  • Check that the ESP32 and sensor share the same GND.
WI-FI

Phone cannot connect

  • Check the SSID and password.
  • Remember that ESP32 Wi-Fi is 2.4 GHz; a 5 GHz-only network will not work.
  • Make sure the phone and ESP32 are on the same local network.
  • Use the exact IP printed by Serial Monitor.
Browser debugging: If JavaScript appears completely dead, open the browser's developer console when available and look for syntax errors or failed network requests.

JavaScript is strict: one missing bracket, quote, parenthesis, or semicolon can cause problems depending on the surrounding code.
13 // VOCABULARY

New Vocabulary

JavaScript
A programming language that runs in the web browser and can manipulate webpage content.
fetch()
A JavaScript API used to make HTTP requests and receive responses asynchronously.
DOM
Document Object Model — the browser's structured representation of the webpage that JavaScript can inspect and modify.
getElementById()
A DOM method used to find an HTML element by its unique ID.
API Endpoint
A server route designed to provide a specific operation or piece of data.
text/plain
An HTTP content type used here for small raw text responses such as sensor numbers or ON/OFF.
Polling
Repeatedly asking a server for fresh information at a chosen interval.
Asynchronous Request
A request that allows other browser work to continue while waiting for the response.
Core idea: We have separated the interface from the data requests . That separation is one of the foundations of modern web applications.
